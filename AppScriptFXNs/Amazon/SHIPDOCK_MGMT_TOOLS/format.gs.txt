/**
 * Formats the 'Format' sheet:
 * 1) Deletes columns A, B, and C.
 * 2) Removes duplicate rows from the remaining data.
 * 3) Filters out (deletes) rows where any value in columns A through J contains 'scKivaA01'.
 */
function formatFormatSheet() {
  const sheetName = 'Format';
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(sheetName);

  if (!sheet) {
    SpreadsheetApp.getUi().alert(`Sheet "${sheetName}" not found.`);
    return;
  }

  // 1) Delete columns A, B, and C (indices 1, 2, and 3)
  // We check if there are at least 3 columns before deleting
  if (sheet.getMaxColumns() >= 3) {
    sheet.deleteColumns(1, 3);
  } else {
    SpreadsheetApp.getUi().alert("The sheet does not have enough columns to delete A, B, and C.");
    return;
  }

  // Get the remaining data to process in memory for better performance
  const range = sheet.getDataRange();
  const values = range.getValues();
  
  if (values.length === 0) {
    SpreadsheetApp.getUi().alert("The sheet is empty after deleting columns.");
    return;
  }

  const targetString = 'scKivaA01';
  const processedData = [];
  const seenRows = new Set();

  for (let i = 0; i < values.length; i++) {
    const row = values[i];

    // 3) Filter out rows where any value in columns A through J contains 'scKivaA01'
    // Note: After deleting A, B, and C, the "remaining columns A-J" are the first 10 columns.
    const checkLimit = Math.min(row.length, 10);
    let containsTarget = false;
    for (let j = 0; j < checkLimit; j++) {
      if (row[j] !== null && String(row[j]).includes(targetString)) {
        containsTarget = true;
        break;
      }
    }

    // If the row contains the target string, we "delete" it by skipping it
    if (containsTarget) continue;

    // 2) Remove duplicate rows
    // We use a string representation of the row as a key to identify duplicates
    const rowKey = JSON.stringify(row);
    if (!seenRows.has(rowKey)) {
      processedData.push(row);
      seenRows.add(rowKey);
    }
  }

  // Clear the existing content and write the cleaned data back to the sheet
  sheet.clearContents();
  if (processedData.length > 0) {
    sheet.getRange(1, 1, processedData.length, processedData[0].length).setValues(processedData);
    SpreadsheetApp.getUi().alert(`Formatting complete! ${values.length - processedData.length} rows were removed.`);
  } else {
    SpreadsheetApp.getUi().alert("Formatting complete! No rows remain after filtering.");
  }
}
