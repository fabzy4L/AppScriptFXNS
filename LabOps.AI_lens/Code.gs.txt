const CLOUD_URL = "https://process-image-434235872268.us-central1.run.app/";

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
      .setTitle('LabMind Lens')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

function processUpload(base64Data) {
  const cleanBase64 = base64Data.split(',')[1];
  
  const payload = { "image_data": cleanBase64 };
  const options = {
    "method": "post",
    "contentType": "application/json",
    "payload": JSON.stringify(payload),
    "muteHttpExceptions": true
  };

  try {
    const response = UrlFetchApp.fetch(CLOUD_URL, options);
    const text = response.getContentText();
    
    // DEBUG LOG: This will show up in your Execution Logs
    console.log("CLOUD SAYS:", text); 
    
    // Parse the JSON
    let result;
    try {
       result = JSON.parse(text);
    } catch (e) {
       return "❌ Error: Cloud returned invalid JSON. Check Logs.";
    }
    
    // THE SAFETY CHECK
    if (result.status === "success" && result.data) {
      logToSheet(result.data);
      // Show me EVERYTHING the AI sent back
      return "✅ Raw Data: " + JSON.stringify(result.data);
    } else {
      // If success is missing or data is missing, tell us why
      return "⚠️ Issue: " + (result.error || "Cloud sent empty data.");
    }
  } catch (e) {
    return "Connection Failed: " + e.toString();
  }
}

function logToSheet(data) {
  // 1. HARDCODE THE ID (Context is often lost in Web Apps)
  var sheetId = "1AU-KwUV6snuX_2_SweNjWv2i3nVi5UfcpcozXRFxCDQ"; 
  
  try {
    var ss = SpreadsheetApp.openById(sheetId);
    var sheet = ss.getSheets()[0]; // Grabs the first tab
    
    // Ensure headers exist
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Item Name", "Category", "ID", "Status", "Details"]);
    }
    
    sheet.appendRow([
      new Date(), 
      data.item_name, 
      data.category, 
      data.identifier, 
      data.status, 
      data.details
    ]);
    
  } catch (e) {
    // If this fails, we log it to the script console so you can debug
    console.error("Sheet Error: " + e.toString());
  }
}
